from __future__ import annotations

class WinProbabilityParams:

        return cls()

    @classmethod
    def for_provider(cls, provider: str) -> "WinProbabilityParams":  # noqa: ARG003 (ADR-009: no tuning yet)
        return cls.default()

```python
# silly_kicks/win_probability/_state.py
from __future__ import annotations
import numpy as np, pandas as pd
from silly_kicks.spadl import config as C
from silly_kicks.id_compat import ids_match
from ._config import WinProbabilityParams

_SHOT_TYPE_IDS = [C.actiontype_id[n] for n in ("shot", "shot_penalty", "shot_freekick")]
_SUCCESS = C.result_id["success"]; _OWNGOAL = C.result_id["owngoal"]
_RED = C.result_id["red_card"]; _YELLOW = C.result_id["yellow_card"]; _FOUL = C.actiontype_id["foul"]
_PERIOD_OFFSET_MIN = {1: 0, 2: 45, 3: 90, 4: 105, 5: 120}  # regulation + ET; PSO excluded upstream

def _home_team_id(actions, games):
    if games is not None:
        return dict(zip(games["game_id"].astype(object), games["home_team_id"], strict=False))
    if "home_team_id" in actions.columns:
        return {g: h for g, h in zip(actions["game_id"], actions["home_team_id"], strict=False)}
    raise ValueError("derive_match_states: home_team_id required via `games` or an actions['home_team_id'] column")

def derive_match_states(actions, *, games=None, params: WinProbabilityParams):
    home_map = _home_team_id(actions, games)
    a = actions.sort_values(["game_id", "period_id", "time_seconds", "action_id"]).reset_index(drop=True)
    # per-row goal credit (id-based, ADR-018): scored shot -> +1 own; owngoal -> +1 opponent
    is_scored_shot = a["type_id"].isin(_SHOT_TYPE_IDS) & (a["result_id"] == _SUCCESS)
    is_owngoal = a["result_id"] == _OWNGOAL
    rows = []
    for game_id, g in a.groupby("game_id", sort=False):
        home_id = home_map.get(game_id)
        teams = list(pd.unique(g["team_id"].dropna()))
        # running (pre-action) tallies keyed by team
        goals = {t: 0 for t in teams}; reds = {t: 0 for t in teams}; yellows = {t: {} for t in teams}
        final_min = _PERIOD_OFFSET_MIN.get(int(g["period_id"].max()), 90) + g.loc[g["period_id"].idxmax(), "time_seconds"]/60.0
        final_min = max(final_min, params.regulation_minutes)
        for _, r in g.iterrows():
            t = r["team_id"]; opp = [x for x in teams if not ids_match(pd.Series([x]), t).iloc[0]]
            opp = opp[0] if opp else None
            abs_min = _PERIOD_OFFSET_MIN.get(int(r["period_id"]), 0) + r["time_seconds"]/60.0
            own_s = goals.get(t, 0); opp_s = goals.get(opp, 0) if opp else 0
            man = reds.get(opp, 0) - reds.get(t, 0) if opp else 0  # opponents sent off - own sent off
            rows.append({
                "game_id": game_id, "action_id": int(r["action_id"]), "period_id": int(r["period_id"]),
                "team_id": t, "absolute_minute": abs_min, "minutes_remaining": max(final_min - abs_min, 0.0),
                "own_score": own_s, "opp_score": opp_s, "score_diff": own_s - opp_s,
                "home": (home_id is not None and bool(ids_match(pd.Series([t]), home_id).iloc[0])),
                "man_advantage": int(man),
                "state_source": "resolved" if (home_id is not None and opp is not None) else "unresolved",
            })
            # advance running tallies AFTER recording the pre-action state
            idx = r.name
            if bool(is_scored_shot.loc[idx]):
                goals[t] = goals.get(t, 0) + 1
            elif bool(is_owngoal.loc[idx]) and opp is not None:
                goals[opp] = goals.get(opp, 0) + 1
            if r["type_id"] == _FOUL and r["result_id"] == _RED:
                reds[t] = reds.get(t, 0) + 1
            elif r["type_id"] == _FOUL and r["result_id"] == _YELLOW:
                pid = r.get("player_id")
                yellows[t][pid] = yellows[t].get(pid, 0) + 1
                if yellows[t][pid] >= 2:
                    reds[t] = reds.get(t, 0) + 1  # second yellow -> red
    return pd.DataFrame(rows).astype({"man_advantage": "int64", "home": "boolean"})
```

## Task 3: `_chain.py` — forward Markov chain + backward-DP `Pwin` table

**Files:**
- Create: `silly_kicks/win_probability/_chain.py`
- Test: `tests/win_probability/test_chain.py`

**Interfaces:**
- Consumes: numpy; `WinProbabilityParams`.
- Produces:
  - `step_matrix(p_home: np.ndarray, p_away: np.ndarray, K: int) -> np.ndarray` — `(2K+1, 2K+1)` row-stochastic transition on `score_diff ∈ [-K, K]`; `p_home[d]`/`p_away[d]` are the per-diff home/away scoring probs for THIS interval. Up (`+1`) w.p. `p_home(1-p_away)`, down (`-1`) w.p. `p_away(1-p_home)`, stay w.p. remainder; edges absorb toward the pad.
  - `win_prob_table(hazard_home, hazard_away, *, n_intervals: int, K: int) -> np.ndarray` — `Pwin` of shape `(2K+1, n_intervals+1)`; `Pwin[d_idx, m]` = `P(final score_diff > 0 | current diff=d, m intervals remain)`, by backward induction. Also returns draw/loss via `outcome_table(...)` or a 3-plane array (choose one; see Step 3).
  - `outcome_from_start(hazard_home, hazard_away, *, start_diff: int, n_intervals: int, K: int) -> tuple[float,float,float]` — a single (win,draw,loss) roll (thin wrapper; special case of the table).
  - `expected_total_goals(hazard_home, hazard_away, *, n_intervals: int, K: int, start_diff: int = 0) -> float` — the expected TOTAL goals over the match. The `score_diff` chain does NOT track total goals (a both-score minute is net-0), so this is the hazard-path integral: forward-propagate the diff distribution and accumulate `Σ_m Σ_d dist_m[d]·(p_home(d,m)+p_away(-d,m))`. ```python
# silly_kicks/win_probability/_chain.py
from __future__ import annotations
import numpy as np

def step_matrix(p_home: np.ndarray, p_away: np.ndarray, K: int) -> np.ndarray:
    n = 2 * K + 1
    up = p_home * (1.0 - p_away)      # score_diff += 1
    dn = p_away * (1.0 - p_home)      # score_diff -= 1
    stay = 1.0 - up - dn
    M = np.zeros((n, n), dtype="float64")
    idx = np.arange(n)
    M[idx, idx] = stay
    M[idx[:-1], idx[:-1] + 1] = up[:-1]         # up
    M[idx[1:], idx[1:] - 1] = dn[1:]            # down
    # absorbing pad edges: mass that would leave stays put (negligible tail; pinned by expected-goals gate)
    M[0, 0] += dn[0]
    M[-1, -1] += up[-1]
    return M

def _diffs_axis(K): return np.arange(-K, K + 1)

def win_prob_table(hazard_home, hazard_away, *, n_intervals: int, K: int) -> np.ndarray:
    """Pwin[d_idx, m] = P(final score_diff > 0 | current diff, m intervals remain). Backward induction."""
    diffs = _diffs_axis(K)
    Pwin = np.zeros((2 * K + 1, n_intervals + 1), dtype="float64")
    Pwin[:, 0] = (diffs > 0).astype("float64")   # terminal: already decided
    for m in range(1, n_intervals + 1):
        # minutes_remaining for this step: m intervals * interval_minutes handled by caller's hazard closure
        ph = np.array([hazard_home(int(d), float(m)) for d in diffs])
        pa = np.array([hazard_away(int(d), float(m)) for d in diffs])
        M = step_matrix(ph, pa, K)
        Pwin[:, m] = M @ Pwin[:, m - 1]
    return Pwin

def _outcome_planes(hazard_home, hazard_away, *, n_intervals: int, K: int):
    diffs = _diffs_axis(K)
    W = (diffs > 0).astype("float64"); D = (diffs == 0).astype("float64"); L = (diffs < 0).astype("float64")
    for m in range(1, n_intervals + 1):
        ph = np.array([hazard_home(int(d), float(m)) for d in diffs])
        pa = np.array([hazard_away(int(d), float(m)) for d in diffs])
        M = step_matrix(ph, pa, K)
        W, D, L = M @ W, M @ D, M @ L
    return W, D, L

def outcome_from_start(hazard_home, hazard_away, *, start_diff: int, n_intervals: int, K: int):
    W, D, L = _outcome_planes(hazard_home, hazard_away, n_intervals=n_intervals, K=K)
    i = start_diff + K
    return float(W[i]), float(D[i]), float(L[i])

def expected_total_goals(hazard_home, hazard_away, *, n_intervals: int, K: int, start_diff: int = 0) -> float:
    """Expected TOTAL goals over the match (hazard-path integral; the diff chain does not track totals)."""
    diffs = _diffs_axis(K)
    dist = np.zeros(2 * K + 1); dist[start_diff + K] = 1.0
    total = 0.0
    for m in range(n_intervals, 0, -1):     # m = intervals remaining at this step
        ph = np.array([hazard_home(int(d), float(m)) for d in diffs])
        pa = np.array([hazard_away(int(d), float(m)) for d in diffs])
        total += float(dist @ (ph + pa))    # expected goals THIS interval under the current diff dist
        dist = step_matrix(ph, pa, K) @ dist
    return total
```

def test_bundled_expected_goals_matches_corpus_rate():
    import json, pathlib
    from silly_kicks.win_probability import WinProbabilityModel, WinProbabilityParams
    from silly_kicks.win_probability._chain import expected_total_goals
    m = WinProbabilityModel.bundled(); p = WinProbabilityParams.default()
    wdir = pathlib.Path(__file__).resolve().parents[2] / "silly_kicks" / "win_probability" / "weights"
    meta = json.loads((wdir / "metadata.json").read_text(encoding="utf-8"))
    corpus_rate = meta["corpus_goals_per_match"]; tol = meta["expected_goals_tol"]
    n = p.regulation_minutes // p.interval_minutes
    hz_home = lambda d, mm: m._hazard(score_diff=d,  minutes_remaining=mm, base_strength=0.0, home=True,  man_advantage=0)
    hz_away = lambda d, mm: m._hazard(score_diff=-d, minutes_remaining=mm, base_strength=0.0, home=False, man_advantage=0)
    eg = expected_total_goals(hz_home, hz_away, n_intervals=n, K=p.lattice_pad, start_diff=0)
    assert abs(eg - corpus_rate) < tol   # fitted-hazard scale matches the corpus; mis-scale fails loud
```
